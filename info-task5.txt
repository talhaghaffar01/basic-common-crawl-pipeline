Run the Worker

With tokenizer training:

python worker.py \
    --minio-endpoint=localhost:9500 \
    --minio-access-key=minioadmin \
    --minio-secret-key=minioadmin \
    --minio-bucket=commoncrawl-data \
    --train-tokenizer \
    --min-length=500 \
    --max-length=1000000

With pre-trained tokenizer:

python worker.py \
    --minio-endpoint=localhost:9500 \
    --minio-access-key=minioadmin \
    --minio-secret-key=minioadmin \
    --minio-bucket=commoncrawl-data \
    --tokenizer-path=tokenizer.json \
    --min-length=500 \
    --max-length=1000000
